package org.compiler;

import java.util.*;

public class Main {
    private static int value;
    
    public Main(int x) {
        value = x;
    }
    
    public int calculate(int a, int b) {
        if (a > b) {
            return a + b;
        } else if (a < b) {
            return a - b;
        } else {
            return a * b;
        }
    }
    
    public void process() {
        try {
            int[] arr = new int[10];
            for (int i = 0; i < arr.length; i++) {
                arr[i] = i * 2;
            }
            while (value > 0) {
                value--;
            }
        } catch (Exception e) {
            System.out.println("Error occurred");
        } finally {
            System.out.println("Cleanup");
        }
    }
}
